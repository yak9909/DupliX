# the editor scripts of tests/ae/dx2_graph.jsx, run by the effect through the editor when Random Seed is set
== seed 1 ==
# comp B: a circle of 6, its Radius exposed (Macro 1: nothing is linked after "clear"), turned 45 degrees, 50%
clear
add Circle 40 60 as c
add Modify 200 60 as m
add Output 360 60 as o
wire c m
wire m o
set c count 6
expose c radius
set m rotZ 45
set m opacity -50
== seed 2 ==
# comp C: the default grid (Macro 1-3 in use) with a second layer in z; Spacing Z exposed lands on Macro 4, a row
# AE hides until the graph uses it (the user's first Expose on a new instance)
set 1 countZ 2
set 1 spacingZ 80
expose 1 spacingZ
== seed 3 ==
# comp D: a circle of 7 not wired. run_dx2_graph.ps1 then drives the editor's window with the mouse: drags from the
# circle's output (260, 291) to the Output's input (330, 101), clicks the circle's Count (200, 312) and types 9
add Circle 40 250 as c
set c count 7
== seed 4 ==
# comp E: what the Overview's Circle button does (the same function), not an editor script
uipreset Circle
